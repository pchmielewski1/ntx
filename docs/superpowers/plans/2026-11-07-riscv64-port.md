- [ ] **Step 1: Utwórz `scripts/riscv/qemu-user-run.sh` (dokładnie ta treść)**

```bash
#!/usr/bin/env bash
#
# Run a dynamically linked riscv64 glibc binary under qemu-riscv64 user-mode
# emulation (x86_64 or aarch64 host).
#
# Usage: scripts/riscv/qemu-user-run.sh <binary> [args...]
#
# <binary> is absolute or relative to the current directory. The guest exit
# code becomes this script's exit code. Environment:
#   QEMU_RISCV64   qemu binary (default: qemu-riscv64, else qemu-riscv64-static, on PATH)
#   RISCV_SYSROOT  glibc sysroot holding the riscv64 dynamic loader
#                  (default: /usr/riscv64-linux-gnu from libc6-riscv64-cross)
set -euo pipefail

if [ "$#" -lt 1 ]; then
    echo "usage: $0 <binary> [args...]" >&2
    exit 2
fi

bin=$1
shift
case "$bin" in
    /*) ;;
    *) bin="$PWD/$bin" ;;
esac
if [ ! -f "$bin" ]; then
    echo "[qemu-riscv64] ERROR: binary not found: $bin" >&2
    exit 127
fi

sysroot="${RISCV_SYSROOT:-/usr/riscv64-linux-gnu}"
if [ ! -e "$sysroot/lib/ld-linux-riscv64-lp64d.so.1" ]; then
    echo "[qemu-riscv64] ERROR: sysroot missing: $sysroot/lib/ld-linux-riscv64-lp64d.so.1 (install libc6-dev-riscv64-cross or set RISCV_SYSROOT)" >&2
    exit 127
fi

qemu="${QEMU_RISCV64:-}"
if [ -z "$qemu" ]; then
    for name in qemu-riscv64 qemu-riscv64-static; do
        if command -v "$name" >/dev/null 2>&1; then
            qemu=$name
            break
        fi
    done
fi
if [ -z "$qemu" ]; then
    echo "[qemu-riscv64] ERROR: no qemu-riscv64 on PATH (install qemu-user or set QEMU_RISCV64)" >&2
    exit 127
fi

export QEMU_LD_PREFIX="$sysroot"
exec "$qemu" -0 ntx "$bin" "$@"
```

- [ ] **Step 3: Utwórz `scripts/riscv/docker-run.sh` (dokładnie ta treść)**

```bash
#!/usr/bin/env bash
#
# Run a command in an Ubuntu 22.04 container that has the riscv64 cross
# toolchain and qemu-user (the same packages as the GitHub Actions riscv64
# jobs), as the calling user, with the repository mounted at /repo.
#
# Usage: scripts/riscv/docker-run.sh <command> [args...]
#   e.g. scripts/riscv/docker-run.sh make riscv test-riscv
#
# The image ntx-riscv64-env:22.04 is built from scripts/riscv/Dockerfile on
# first use; `docker rmi ntx-riscv64-env:22.04` forces a rebuild. The platform
# is pinned to the host's so a cached foreign-arch ubuntu image is never used,
# and the container runs as the calling uid so no root-owned files land in
# the repository.
set -euo pipefail

cd "$(dirname "$0")/../.."

if [ "$#" -lt 1 ]; then
    echo "usage: $0 <command> [args...]" >&2
    exit 2
fi
if ! command -v docker >/dev/null 2>&1; then
    echo "[docker-run] ERROR: docker not found on PATH" >&2
    exit 127
fi

case "$(uname -m)" in
    x86_64) platform=linux/amd64 ;;
    aarch64|arm64) platform=linux/arm64 ;;
    *) echo "[docker-run] ERROR: unsupported host architecture $(uname -m)" >&2; exit 2 ;;
esac

image=ntx-riscv64-env:22.04
if ! docker image inspect "$image" >/dev/null 2>&1; then
    docker build --platform "$platform" -t "$image" scripts/riscv
fi

exec docker run --rm --platform "$platform" --user "$(id -u):$(id -g)" -e HOME=/tmp \
    -e NTX_VERSION -e NTX_TEST_TIMEOUT_SCALE -e NTX_TEST_TIMEOUT -e NTX_TEST_GLOB \
    -v "$PWD:/repo" -w /repo "$image" "$@"
```

Run: `chmod +x scripts/riscv/docker-run.sh`

- [ ] **Step 1: Utwórz `scripts/riscv/run-suite.sh` (dokładnie ta treść)**

```bash
#!/usr/bin/env bash
#
# Cross-compile every test/t_*.c with the riscv64 toolchain and run it under
# qemu-riscv64 user-mode emulation.
#
# Usage (from anywhere): scripts/riscv/run-suite.sh
#
# Each test is built with the same flags as `make test` (dynamic glibc) to
# test/.riscv64.<base>.tmp (gitignored), run from the repo root through
# scripts/riscv/qemu-user-run.sh, and removed. Failures accumulate; the list is
# printed at the end and the script exits nonzero if any test failed, or
# prints ALL PASS and exits 0 when the whole suite is green.
#
# NTX_TEST_TIMEOUT_SCALE is exported to the tests (test/tscale.h).
# NTX_TEST_TIMEOUT (default 300 s) bounds each test, so a hang fails instead of
# blocking; exit 124 (TERM) or 137 (KILL after -k 10) is a timeout failure.
# NTX_TEST_GLOB overrides the test file glob (debugging aid).
set -uo pipefail

cd "$(dirname "$0")/../.."

if [ ! -d test ] || [ ! -f scripts/riscv/qemu-user-run.sh ]; then
    echo "[run-suite] ERROR: must run from the nanotorrent repo root" >&2
    exit 2
fi

RISCV_CC="${RISCV_CC:-riscv64-linux-gnu-gcc}"
CFLAGS="${CFLAGS:--O2 -std=c11 -Wall -Wextra -Werror -D_POSIX_C_SOURCE=200809L -ffunction-sections}"
LDFLAGS="${LDFLAGS:--Wl,--gc-sections}"
LDLIBS="${LDLIBS:--lm}"
export NTX_TEST_TIMEOUT_SCALE="${NTX_TEST_TIMEOUT_SCALE:-1}"
NTX_TEST_TIMEOUT="${NTX_TEST_TIMEOUT:-300}"
TEST_GLOB="${NTX_TEST_GLOB:-test/t_*.c}"

if ! command -v timeout >/dev/null 2>&1; then
    echo "[run-suite] ERROR: coreutils 'timeout' not found on PATH (hang guard requires it)" >&2
    exit 127
fi
if ! command -v "$RISCV_CC" >/dev/null 2>&1; then
    echo "[run-suite] ERROR: RISCV_CC not found: $RISCV_CC (install gcc-riscv64-linux-gnu)" >&2
    exit 127
fi

failures=()
total=0

shopt -s nullglob
test_files=($TEST_GLOB)
shopt -u nullglob

if [ "${#test_files[@]}" -eq 0 ]; then
    echo "[run-suite] ERROR: test glob '$TEST_GLOB' matched zero test files — refusing to report ALL PASS" >&2
    exit 1
fi

for f in "${test_files[@]}"; do
    base="$(basename "$f" .c)"
    bin="test/.riscv64.${base}.tmp"
    total=$((total + 1))
    printf '==> %s ' "$base"

    # shellcheck disable=SC2086 # CFLAGS/LDFLAGS/LDLIBS are word-split by design
    if ! "$RISCV_CC" $CFLAGS -o "$bin" "$f" $LDFLAGS $LDLIBS; then
        echo "COMPILE FAIL"
        failures+=("$base (compile)")
        rm -f "$bin"
        continue
    fi

    timeout -k 10 "$NTX_TEST_TIMEOUT" bash scripts/riscv/qemu-user-run.sh "$bin"
    rc=$?
    rm -f "$bin"
    if [ "$rc" -eq 124 ] || [ "$rc" -eq 137 ]; then
        echo "TIMEOUT (exceeded ${NTX_TEST_TIMEOUT}s)"
        failures+=("$base (timeout ${NTX_TEST_TIMEOUT}s rc=$rc)")
    elif [ "$rc" -ne 0 ]; then
        echo "RUN FAIL (rc=$rc)"
        failures+=("$base (run rc=$rc)")
    else
        echo "ok"
    fi
done

echo
echo "suite: $total tests, $(( total - ${#failures[@]} )) passed, ${#failures[@]} failed"
if [ "${#failures[@]}" -gt 0 ]; then
    echo "FAILURES:"
    for x in "${failures[@]}"; do
        echo "  FAIL $x"
    done
    exit 1
fi
echo "ALL PASS"
exit 0
```

```bash
#!/usr/bin/env bash
#
# Run the shell-level gates (test-cli, test-net, test-ipc, test-tls13) against
# the riscv64 binary under qemu-riscv64 user-mode emulation.
#
# Usage (from anywhere, after `make riscv`): scripts/riscv/run-shell-suites.sh
#
# The gates hard-code ./ntx relative to their own repo root, so this script
# builds a scratch root test/.scratch/riscv64-shell/ holding copies of the gate
# scripts and an executable ./ntx wrapper that runs the riscv64 binary through
# scripts/riscv/qemu-user-run.sh. The native ./ntx is never read, built or
# written. The scratch root is removed when every gate passes and kept for
# inspection otherwise.
#
# NTX_RISCV_BIN overrides the binary under test (default: ntx-riscv64).
# test-tls13 exiting 77 (no openssl / cannot listen) is a skip, as in
# `make test-tls13`.
set -uo pipefail

cd "$(dirname "$0")/../.."
REPO=$PWD
RISCV_CC="${RISCV_CC:-riscv64-linux-gnu-gcc}"
BIN="${NTX_RISCV_BIN:-ntx-riscv64}"
case "$BIN" in
    /*) ;;
    *) BIN="$REPO/$BIN" ;;
esac

if [ ! -f "$BIN" ]; then
    echo "[riscv-shell] ERROR: $BIN not found (run make riscv first)" >&2
    exit 2
fi
# ELF e_machine (2 bytes at offset 18, little-endian) must be 0x00f3 = EM_RISCV.
if [ "$(od -An -tx1 -j18 -N2 "$BIN" | tr -d ' \n')" != "f300" ]; then
    echo "[riscv-shell] ERROR: $BIN is not a RISC-V ELF binary" >&2
    exit 2
fi

W="$REPO/test/.scratch/riscv64-shell"
rm -rf "$W"
mkdir -p "$W/test/interop/ipc"
cp test/cli_errors.sh test/net_loopback.sh test/tls13_interop.sh "$W/test/"
cp test/interop/ipc/interop_ipc.sh test/interop/ipc/assert_ipc.py "$W/test/interop/ipc/"

wrap() { # wrap <wrapper-path> <riscv64-binary>
    printf '#!/bin/sh\nexec bash "%s/scripts/riscv/qemu-user-run.sh" "%s" "$@"\n' "$REPO" "$2" > "$1"
    chmod +x "$1"
}
wrap "$W/ntx" "$BIN"

# Same sources and flags test/tls13_interop.sh uses for its native probe.
TLS_SRC=$(sed -n 's/^  SRC="\(.*\)"$/\1/p' test/tls13_interop.sh)
if [ -z "$TLS_SRC" ]; then
    echo "[riscv-shell] ERROR: cannot read the SRC= list from test/tls13_interop.sh" >&2
    exit 2
fi
# shellcheck disable=SC2086 # TLS_SRC is a word list by design
if ! "$RISCV_CC" -O2 -std=c11 -Wall -Wextra -D_POSIX_C_SOURCE=200809L -D_DEFAULT_SOURCE -DNTX_TLS13_LIVE=1 \
        -o "$W/tls_probe-riscv64" test/tls_probe.c $TLS_SRC -lm; then
    echo "[riscv-shell] ERROR: tls_probe cross-compile failed" >&2
    exit 1
fi
wrap "$W/tls_probe" "$W/tls_probe-riscv64"

failures=()
gate() { # gate <name> <command...>
    local name=$1 rc
    shift
    echo "==> $name"
    "$@"
    rc=$?
    if [ "$rc" -eq 0 ] || { [ "$name" = test-tls13 ] && [ "$rc" -eq 77 ]; }; then
        echo "ok $name (rc=$rc)"
    else
        echo "FAIL $name (rc=$rc)"
        failures+=("$name")
    fi
}

gate test-cli   sh   "$W/test/cli_errors.sh"
gate test-net   sh   "$W/test/net_loopback.sh"
gate test-ipc   bash "$W/test/interop/ipc/interop_ipc.sh"
gate test-tls13 bash "$W/test/tls13_interop.sh" "$W/tls_probe"

echo
if [ "${#failures[@]}" -gt 0 ]; then
    echo "riscv64 shell gates FAILED: ${failures[*]} (scratch kept: $W)"
    exit 1
fi
rm -rf "$W"
echo "ALL PASS (riscv64 shell gates)"
```

      - name: Build

      - name: Check the version stamp
        run: |
          out=$(bash scripts/riscv/qemu-user-run.sh ntx-riscv64 --version)
          echo "$out"
          [ "$out" = "ntx $NTX_RELEASE_VERSION" ]

      - name: Package
        run: |
          set -euo pipefail
          mkdir dist
          cp ntx-riscv64 "dist/ntx-${NTX_RELEASE_VERSION}-linux-riscv64"
          riscv64-linux-gnu-strip --strip-unneeded "dist/ntx-${NTX_RELEASE_VERSION}-linux-riscv64"

          name: dist-riscv64
          path: dist/*
          if-no-files-found: error
